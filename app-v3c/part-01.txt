  function rotate(x,y,a) { const c=Math.cos(a),s=Math.sin(a); return [x*c-y*s,x*s+y*c]; }

  function pointSegmentMeters(p,a,b){const lat=(a[1]+b[1]+p[1])/3*Math.PI/180,kx=111320*Math.cos(lat),ky=110540; const ax=a[0]*kx,ay=a[1]*ky,bx=b[0]*kx,by=b[1]*ky,px=p[0]*kx,py=p[1]*ky; const dx=bx-ax,dy=by-ay,t=Math.max(0,Math.min(1,((px-ax)*dx+(py-ay)*dy)/((dx*dx+dy*dy)||1))); return Math.hypot(px-(ax+t*dx),py-(ay+t*dy)); }
