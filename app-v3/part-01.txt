  async function parsePack(buffer) {
    const c = JSON.parse(await decodePack(buffer));
    if (c.format !== 'disaster-map-pack-container' || !c.manifest?.id) throw new Error('地域パック形式が不正です。');
    const files = [];
    for (const def of c.manifest.files || []) {
      const p = c.files?.[def.path]; if (!p) throw new Error(`欠落: ${def.path}`);
      const bytes = p.encoding === 'base64' ? b64(p.data) : new TextEncoder().encode(p.data);
      const hash = await sha256Hex(bytes); if (def.sha256 && hash !== def.sha256) throw new Error(`SHA-256不一致: ${def.path}`);
      files.push({ key:`${c.manifest.id}::${def.path}`, packId:c.manifest.id, path:def.path, role:def.role, mediaType:def.mediaType, blob:new Blob([bytes],{type:def.mediaType}), sha256:hash });
    }
    return { manifest:c.manifest, files };
  }
